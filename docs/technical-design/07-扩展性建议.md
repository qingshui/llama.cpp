# llama.cpp 技术设计文档

## 第七部分：扩展性建议

### 7.1 扩展性概述

llama.cpp 的架构设计具有良好的扩展性，支持以下维度的扩展：

```
┌─────────────────────────────────────────────────────────────┐
│                    模型架构扩展                              │
│  - 添加新模型支持                                            │
│  - 扩展模型组件（注意力、FFN、归一化）                        │
│  - 支持新的激活函数                                          │
├─────────────────────────────────────────────────────────────┤
│                    后端扩展                                  │
│  - 添加新的硬件后端                                          │
│  - 实现新的量化方法                                          │
│  - 优化现有后端实现                                          │
├─────────────────────────────────────────────────────────────┤
│                    功能扩展                                  │
│  - 添加新的采样器                                            │
│  - 扩展 API 接口                                              │
│  - 添加新的工具和功能                                        │
├─────────────────────────────────────────────────────────────┤
│                    系统集成扩展                              │
│  - 嵌入到现有系统                                            │
│  - 构建分布式推理系统                                        │
│  - 与第三方框架集成                                          │
└─────────────────────────────────────────────────────────────┘
```

```cpp
// src/models/my_model.cpp

#include "llama-arch.h"
#include "llama-model.h"
#include "ggml.h"

struct my_model_hparams {
    uint32_t n_vocab;
    uint32_t n_embd;
    uint32_t n_layer;
    uint32_t n_head;
    uint32_t n_head_kv;
    float rope_freq_base;
    float norm_eps;
    // ... 其他超参数 ...
};

struct my_model_layer {
    struct ggml_tensor * wq;
    struct ggml_tensor * wk;
    struct ggml_tensor * wv;
    struct ggml_tensor * wo;
    struct ggml_tensor * wffn_up;
    struct ggml_tensor * wffn_down;
    struct ggml_tensor * wffn_gate;
    struct ggml_tensor * attn_norm;
    struct ggml_tensor * ffn_norm;
};

// 加载模型权重
static bool my_model_load_tensors(
    llama_model_loader & loader,
    llama_model & model)
{
    auto & hparams = model.hparams;
    auto & layers = model.layers;

    // 加载嵌入
    model.t_input = loader.get_tensor("token_embd", {hparams.n_embd, hparams.n_vocab});

    // 加载各层
    for (int i = 0; i < hparams.n_layer; i++) {
        auto & layer = layers[i];

        // 注意力权重
        layer.wq = loader.get_tensor(format("blk.%d.attn_q", i),
                                     {hparams.n_embd, hparams.n_embd});
        layer.wk = loader.get_tensor(format("blk.%d.attn_k", i),
                                     {hparams.n_embd, hparams.n_head_kv * hparams.n_embd_head_k});
        layer.wv = loader.get_tensor(format("blk.%d.attn_v", i),
                                     {hparams.n_embd, hparams.n_head_kv * hparams.n_embd_head_v});
        layer.wo = loader.get_tensor(format("blk.%d.attn_o", i),
                                     {hparams.n_embd, hparams.n_embd});

        // FFN 权重
        layer.wffn_up = loader.get_tensor(format("blk.%d.ffn_up", i),
                                          {hparams.n_embd, hparams.n_ffn});
        layer.wffn_down = loader.get_tensor(format("blk.%d.ffn_down", i),
                                            {hparams.n_ffn, hparams.n_embd});
        layer.wffn_gate = loader.get_tensor(format("blk.%d.ffn_gate", i),
                                            {hparams.n_embd, hparams.n_ffn});

        // 归一化
        layer.attn_norm = loader.get_tensor(format("blk.%d.attn_norm", i),
                                            {hparams.n_embd});
        layer.ffn_norm = loader.get_tensor(format("blk.%d.ffn_norm", i),
                                           {hparams.n_embd});
    }

    // 加载输出层
    model.t_output_norm = loader.get_tensor("output_norm", {hparams.n_embd});
    model.t_output = loader.get_tensor("output", {hparams.n_embd, hparams.n_vocab});

    return true;
}
```

**步骤 3：实现前向传播**

```cpp
// src/models/my_model.cpp

struct ggml_tensor * my_model_forward(
    llama_context & lctx,
    llama_batch & batch,
    int il)  // 层索引
{
    auto & model = lctx.model;
    auto & hparams = model.hparams;
    auto & cparams = lctx.cparams;
    auto & layer = model.layers[il];

    // 输入张量
    struct ggml_tensor * cur = get_input_tensor(lctx, batch);

    // 注意力归一化
    cur = ggml_rms_norm(lctx.ctx, cur, hparams.norm_eps);
    cur = ggml_mul(lctx.ctx, cur, layer.attn_norm);

    // 计算 Q, K, V
    struct ggml_tensor * Q = ggml_mul_mat(lctx.ctx, layer.wq, cur);
    struct ggml_tensor * K = ggml_mul_mat(lctx.ctx, layer.wk, cur);
    struct ggml_tensor * V = ggml_mul_mat(lctx.ctx, layer.wv, cur);

    // 重塑为多头
    Q = ggml_reshape_4d(lctx.ctx, Q,
                        hparams.n_embd_head, hparams.n_head,
                        batch.n_tokens, 1);
    K = ggml_reshape_4d(lctx.ctx, K,
                        hparams.n_embd_head, hparams.n_head_kv,
                        batch.n_tokens, 1);
    V = ggml_reshape_4d(lctx.ctx, V,
                        hparams.n_embd_head, hparams.n_head_kv,
                        batch.n_tokens, 1);

    // 旋转位置编码 (RoPE)
    Q = ggml_rope(lctx.ctx, Q, lctx.inp_pos, rope_params);
    K = ggml_rope(lctx.ctx, K, lctx.inp_pos, rope_params);

    // 存储 KV 缓存
    ggml_build_forward_expand(&lctx.graph, ggml_copy(lctx.ctx, K, lctx.kv_self.k[il]));
    ggml_build_forward_expand(&lctx.graph, ggml_copy(lctx.ctx, V, lctx.kv_self.v[il]));

    // 注意力计算
    struct ggml_tensor * KQ = ggml_mul_mat(lctx.ctx, K, Q);
    struct ggml_tensor * KQ_scaled = ggml_scale(lctx.ctx, KQ, 1.0f / sqrt(hparams.n_embd_head));
    struct ggml_tensor * KQ_masked = ggml_add(lctx.ctx, KQ_scaled, lctx.inp_KQ_mask);
    struct ggml_tensor * KQ_soft_max = ggml_soft_max(lctx.ctx, KQ_masked);

    struct ggml_tensor * KQV = ggml_mul_mat(lctx.ctx, V, KQ_soft_max);

    // 重塑回嵌入维度
    KQV = ggml_reshape_3d(lctx.ctx, KQV,
                          hparams.n_embd, batch.n_tokens, 1);

    // 输出投影
    cur = ggml_mul_mat(lctx.ctx, layer.wo, KQV);

    // 残差连接
    struct ggml_tensor * inpL = get_input_tensor(lctx, batch);
    cur = ggml_add(lctx.ctx, cur, inpL);

    // FFN 归一化
    cur = ggml_rms_norm(lctx.ctx, cur, hparams.norm_eps);
    cur = ggml_mul(lctx.ctx, cur, layer.ffn_norm);

    // FFN 计算 (SwiGLU)
    struct ggml_tensor * FFN_up = ggml_mul_mat(lctx.ctx, layer.wffn_up, cur);
    struct ggml_tensor * FFN_gate = ggml_mul_mat(lctx.ctx, layer.wffn_gate, cur);
    FFN_gate = ggml_silu(lctx.ctx, FFN_gate);
    struct ggml_tensor * FFN = ggml_mul(lctx.ctx, FFN_up, FFN_gate);
    struct ggml_tensor * FFN_down = ggml_mul_mat(lctx.ctx, layer.wffn_down, FFN);

    // 残差连接
    cur = ggml_add(lctx.ctx, cur, FFN_down);

    return cur;
}
```

```cpp
static void my_sampler_accept(llama_sampler * smpl, llama_token token) {
    // 接受 token 时的处理（如更新历史）
    my_sampler_context * ctx = (my_sampler_context *)smpl->ctx;
    // ... 更新状态 ...
}

static void my_sampler_apply(llama_sampler * smpl, llama_token_data_array * cur_p) {
    my_sampler_context * ctx = (my_sampler_context *)smpl->ctx;

    // 应用采样逻辑
    // 1. 过滤低于阈值的 token
    for (size_t i = 0; i < cur_p->size; i++) {
        if (cur_p->data[i].p < ctx->threshold) {
            cur_p->data[i].p = 0;
        }
    }

    // 2. 重新归一化
    float sum = 0;
    for (size_t i = 0; i < cur_p->size; i++) {
        sum += cur_p->data[i].p;
    }
    if (sum > 0) {
        for (size_t i = 0; i < cur_p->size; i++) {
            cur_p->data[i].p /= sum;
        }
    }

    // 3. 排序（可选）
    std::sort(cur_p->data, cur_p->data + cur_p->size,
              [](const llama_token_data & a, const llama_token_data & b) {
                  return a.p > b.p;
              });
    cur_p->sorted = true;
}

static void my_sampler_reset(llama_sampler * smpl) {
    // 重置采样器状态
    my_sampler_context * ctx = (my_sampler_context *)smpl->ctx;
    // ... 重置状态 ...
}

static llama_sampler * my_sampler_clone(const llama_sampler * smpl) {
    // 克隆采样器
    my_sampler_context * old_ctx = (my_sampler_context *)smpl->ctx;
    my_sampler_context * new_ctx = new my_sampler_context{*old_ctx};
```

#### 7.5.2 C++ API 包装

```cpp
// include/llama-cpp.h

#include "llama.h"
#include <memory>
#include <string>
#include <vector>

namespace llama {

// 模型类
class Model {
public:
    static Model load_from_file(
        const std::string & path,
        const ModelParams & params = {})
    {
        llama_model_params c_params = llama_model_default_params();
        // 转换参数
        c_params.n_gpu_layers = params.n_gpu_layers;
        // ...

        llama_model * c_model = llama_load_model_from_file(
            path.c_str(), c_params);

        if (!c_model) {
            throw std::runtime_error("Failed to load model");
        }

        return Model(c_model);
    }

    ~Model() {
        llama_free_model(model_);
    }

    // 禁止拷贝
    Model(const Model &) = delete;
    Model & operator=(const Model &) = delete;

    // 允许移动
    Model(Model && other) noexcept : model_(other.model_) {
        other.model_ = nullptr;
    }

    // 获取信息
    size_t n_vocab() const { return llama_model_n_vocab(model_); }
    size_t n_embd() const { return llama_model_n_embd(model_); }
    size_t n_layer() const { return llama_model_n_layer(model_); }

    llama_model * get() const { return model_; }

private:
    explicit Model(llama_model * model) : model_(model) {}
    llama_model * model_;
};

// 上下文类
class Context {
public:
    static Context from_model(
        const Model & model,
        const ContextParams & params = {})
    {
        llama_context_params c_params = llama_context_default_params();
        // 转换参数
        c_params.n_ctx = params.n_ctx;
        c_params.n_batch = params.n_batch;
        // ...

        llama_context * c_ctx = llama_init_from_model(model.get(), c_params);

        if (!c_ctx) {
            throw std::runtime_error("Failed to create context");
        }

        return Context(c_ctx);
    }

    ~Context() {
        llama_free(ctx_);
    }

    // 解码
    void decode(const Batch & batch) {
        int32_t result = llama_decode(ctx_, batch.get());
        if (result < 0) {
            throw std::runtime_error("llama_decode failed");
        }
    }

    // 获取 logits
    std::vector<float> get_logits() const {
        float * logits = llama_get_logits(ctx_);
        size_t n_vocab = llama_model_n_vocab(llama_get_model(ctx_));
        return std::vector<float>(logits, logits + n_vocab);
    }

    llama_context * get() const { return ctx_; }

private:
    explicit Context(llama_context * ctx) : ctx_(ctx) {}
    llama_context * ctx_;
};

// 批处理类
class Batch {
public:
    Batch(size_t n_tokens, size_t n_seq_max = 1) {
        batch_ = llama_batch_init(n_tokens, 0, n_seq_max);
    }

    ~Batch() {
        llama_batch_free(batch_);
    }

    void add_token(
        llama_token token,
        llama_pos pos,
        const std::vector<llama_seq_id> & seq_ids)
    {
        llama_batch_add(batch_, token, pos, seq_ids);
    }

    llama_batch get() const { return batch_; }

private:
    llama_batch batch_;
};

// 采样器类
class Sampler {
public:
    Sampler(const Model & model, const SamplingParams & params) {
        common_params_sampling c_params;
        // 转换参数
        c_params.temp = params.temperature;
        c_params.top_k = params.top_k;
        c_params.top_p = params.top_p;
        // ...

        sampler_ = common_sampler_init(model.get(), c_params);
    }

    ~Sampler() {
        common_sampler_free(sampler_);
    }

    llama_token sample(Context & ctx, int idx = -1) {
        return common_sampler_sample(sampler_, ctx.get(), idx);
    }

    void accept(llama_token token) {
        common_sampler_accept(sampler_, token, true);
    }

private:
    common_sampler * sampler_;
};

} // namespace llama
```

```cpp
class LlamaWrapper {

    // 流式回调

    void generate_stream(
        const std::string & prompt,
        int max_tokens,
        StreamCallback callback);

    // 嵌入生成
    std::vector<float> get_embedding(const std::string & text);

private:
    struct Impl;
    std::unique_ptr<Impl> impl_;
```

```cpp
// tests/test-my-model.cpp

#include "llama.h"
#include <cassert>
#include <cmath>

void test_my_model_load() {
    // 测试模型加载
    llama_model_params params = llama_model_default_params();
    params.n_gpu_layers = 0;  // CPU only

    llama_model * model = llama_load_model_from_file(
        "models/my-model.Q4_K_M.gguf", params);

    assert(model != nullptr);
    assert(llama_model_n_vocab(model) == 32000);
    assert(llama_model_n_embd(model) == 4096);
    assert(llama_model_n_layer(model) == 32);

    llama_free_model(model);
}

void test_my_model_inference() {
    // 测试推理
    llama_model * model = load_test_model();
    llama_context_params ctx_params = llama_context_default_params();
    ctx_params.n_ctx = 512;

    llama_context * ctx = llama_init_from_model(model, ctx_params);
    assert(ctx != nullptr);

    // 创建批处理
    std::vector<llama_token> tokens = {1, 2, 3, 4, 5};
    llama_batch batch = llama_batch_init(tokens.size(), 0, 1);
    for (size_t i = 0; i < tokens.size(); i++) {
        batch.token[i] = tokens[i];
        batch.pos[i] = i;
        batch.n_seq_id[i] = 1;
        batch.seq_id[i][0] = 0;
        batch.logits[i] = false;
    }
    batch.n_tokens = tokens.size();

    // 执行推理
    int32_t result = llama_decode(ctx, batch);
    assert(result == 0);

    // 检查输出
    float * logits = llama_get_logits(ctx);
    assert(logits != nullptr);

    // 检查 NaN/Inf
    size_t n_vocab = llama_model_n_vocab(model);
    for (size_t i = 0; i < n_vocab; i++) {
        assert(std::isfinite(logits[i]));
    }

    llama_batch_free(batch);
    llama_free(ctx);
    llama_free_model(model);
}

int main() {
    test_my_model_load();
    test_my_model_inference();
    printf("All tests passed!\n");
    return 0;
}
```
