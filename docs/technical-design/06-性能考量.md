#### 6.2.2 量化实现原理

```cpp
// ggml/src/ggml-quants.c

// Q4_0 量化（4-bit，每组 32 个权重）
typedef struct {
    ggml_half d;           // 缩放因子
    uint8_t qs[QK4_0/2];   // 量化权重（4-bit 打包）
} block_q4_0;

#define QK4_0 32

// 量化过程
void quantize_q4_0(const float * weights, block_q4_0 * block) {
    // 1. 计算绝对值最大值作为缩放因子
    float max = 0;
    for (int i = 0; i < QK4_0; i++) {
        max = fmaxf(max, fabsf(weights[i]));
    }
    block->d = ggml_half_from_float(max / 7.0f);

    // 2. 量化每个权重到 4-bit (-8 到 7)
    float id = 1.0f / (max / 7.0f + 1e-6f);
    for (int i = 0; i < QK4_0/2; i++) {
        uint8_t q0 = quantize_to_4bit(weights[i*2] * id);
        uint8_t q1 = quantize_to_4bit(weights[i*2+1] * id);
        block->qs[i] = q0 | (q1 << 4);  // 打包两个 4-bit 值
    }
}

// 反量化过程（用于计算）
void dequantize_q4_0(const block_q4_0 * block, float * weights) {
    float d = ggml_half_to_float(block->d);
    for (int i = 0; i < QK4_0/2; i++) {
        uint8_t q = block->qs[i];
        weights[i*2]   = d * (q & 0x0F) - 8;  // 低 4-bit
        weights[i*2+1] = d * (q >> 4) - 8;    // 高 4-bit
    }
}
```

```cpp
// ARM NEON 向量加法
void ggml_vec_add_f32_neon(int n, float * z, const float * x, const float * y) {
    int i = 0;
    for (; i + 7 < n; i += 8) {
        float32x4_t vx0 = vld1q_f32(x + i);
        float32x4_t vx1 = vld1q_f32(x + i + 4);
        float32x4_t vy0 = vld1q_f32(y + i);
        float32x4_t vy1 = vld1q_f32(y + i + 4);

        float32x4_t vz0 = vaddq_f32(vx0, vy0);
        float32x4_t vz1 = vaddq_f32(vx1, vy1);

        vst1q_f32(z + i, vz0);
        vst1q_f32(z + i + 4, vz1);
    }
    // 处理剩余元素
    for (; i < n; i++) {
        z[i] = x[i] + y[i];
    }
}

// ARM NEON 点积优化
float ggml_vec_dot_f32_neon(int n, const float * x, const float * y) {
    float32x4_t acc = vdupq_n_f32(0);

    for (int i = 0; i < n; i += 4) {
        float32x4_t vx = vld1q_f32(x + i);
        float32x4_t vy = vld1q_f32(y + i);
        acc = vmlaq_f32(acc, vx, vy);  // 融合乘加
    }

    // 水平求和
    return vaddvq_f32(acc);
}
```

```cpp
// ggml/src/ggml-cpu/amx/amx.cpp

// Intel AMX 矩阵乘法优化
void ggml_compute_forward_mul_mat_amx(
    const struct ggml_tensor * src0,
    const struct ggml_tensor * src1,
    struct ggml_tensor * dst)
{
    // AMX 配置
    __tile_config config = {
        .rows = 16,
        .cols = 32,
        .palette = 0
    };

    // 加载配置
    _mm_loadu_si512(&config);
    _tile_loadconfig(&config);

    // 分块计算
    for (int i = 0; i < M; i += 16) {
        for (int j = 0; j < K; j += 32) {
            // 初始化累加器 tile
            _tile_zero(0);

            // 矩阵乘法
            for (int k = 0; k < N; k += 64) {
                // 加载 A 块到 tile
                _tile_loadb(1, src0 + i*n + k);

                // 加载 B 块到 tile
                _tile_loadb(2, src1 + j*k + k);

                // 执行矩阵乘法
                _tile_matmul(0, 1, 2);
            }

            // 存储结果
            _tile_store(0, dst + i*K + j);
        }
    }
}
```

```cpp
// ggml/src/ggml-cuda/mmq.cu

// CUDA 矩阵乘法核函数
__global__ void ggml_cuda_mul_mat_f16(
    const half * a,
    const half * b,
    float * c,
    int M, int N, int K)
{
    // 共享内存
    __shared__ half a_tile[BM][BK];
    __shared__ half b_tile[BK][BN];

    int tx = threadIdx.x;
    int ty = threadIdx.y;

    int row = blockIdx.y * BM + ty;
    int col = blockIdx.x * BN + tx;

    float acc = 0.0f;

    // 分块循环
    for (int t = 0; t < (K + BK - 1) / BK; t++) {
        // 加载 A 块到共享内存
        if (row < M && t*BK + tx < K) {
            a_tile[ty][tx] = a[row * K + t*BK + tx];
        } else {
            a_tile[ty][tx] = 0;
        }

        // 加载 B 块到共享内存（转置加载）
        if (t*BK + ty < K && col < N) {
            b_tile[tx][ty] = b[(t*BK + ty) * N + col];
        } else {
            b_tile[tx][ty] = 0;
        }

        __syncthreads();

        // 计算点积
        #pragma unroll
        for (int k = 0; k < BK; k++) {
            acc += __half2float(a_tile[ty][k]) * __half2float(b_tile[k][tx]);
        }

        __syncthreads();
    }

    // 存储结果
    if (row < M && col < N) {
        c[row * N + col] = acc;
    }
}

// 启动配置
void launch_cuda_mul_mat(...) {
    dim3 blockDim(BN, BM);
    dim3 gridDim((N + BN - 1) / BN, (M + BM - 1) / BM);

    ggml_cuda_mul_mat_f16<<<gridDim, blockDim>>>(a, b, c, M, N, K);
}
```

```cpp
// ggml/src/ggml-metal/ggml-metal.metal

// Metal 矩阵乘法核函数
kernel void ggml_mul_mat_f16(
    device const half * a [[buffer(0)]],
    device const half * b [[buffer(1)]],
    device float * c [[buffer(2)]],
    constant int & M [[buffer(3)]],
    constant int & N [[buffer(4)]],
    constant int & K [[buffer(5)]],
    uint3 gid [[threadgroup_position_in_grid]],
    uint3 tid [[thread_position_in_threadgroup]])
{
    // 线程组共享内存
    threadgroup half a_tile[BM][BK];
    threadgroup half b_tile[BK][BN];

    uint row = gid.y * BM + tid.y;
    uint col = gid.x * BN + tid.x;

    float acc = 0.0f;

    for (int t = 0; t < (K + BK - 1) / BK; t++) {
        // 加载 A 块
        if (row < M && t*BK + tid.x < K) {
            a_tile[tid.y][tid.x] = a[row * K + t*BK + tid.x];
        }

        // 加载 B 块
        if (t*BK + tid.y < K && col < N) {
            b_tile[tid.x][tid.y] = b[(t*BK + tid.y) * N + col];
        }

        threadgroup_barrier(mem_flags::mem_threadgroup);

        // 计算
        for (int k = 0; k < BK; k++) {
            acc += float(a_tile[tid.y][k]) * float(b_tile[k][tid.x]);
        }

        threadgroup_barrier(mem_flags::mem_threadgroup);
    }

    if (row < M && col < N) {
        c[row * N + col] = acc;
    }
}
```

#### 6.4.3 Flash Attention 优化

```cpp
// ggml/src/ggml-cuda/fattn.cu

// Flash Attention 核函数（简化版）
__global__ void flash_attn_f16(
    const half * Q,  // [N, d]
    const half * K,  // [M, d]
    const half * V,  // [M, d']
    float * O,       // [N, d']
    int N, int M, int d, int d_prime)
{
    // 每个线程块处理一个 query 块
    extern __shared__ half shared[];
    half * K_tile = shared;
    half * V_tile = shared + BK * d;

    int n = blockIdx.x * BN + threadIdx.x;
    if (n >= N) return;

    // 加载 Q 向量
    float q[d];
    for (int i = 0; i < d; i++) {
        q[i] = __half2float(Q[n * d + i]);
    }

    float max_val = -INFINITY;
    float sum_exp = 0;
    float acc[d_prime] = {0};

    // 分块处理 K 和 V
    for (int t = 0; t < (M + BK - 1) / BK; t++) {
        // 加载 K 块到共享内存
        for (int i = threadIdx.x; i < BK * d; i += blockDim.x) {
            int m = t * BK + i / d;
            if (m < M) {
                K_tile[i] = K[m * d + (i % d)];
            }
        }

        // 加载 V 块到共享内存
        for (int i = threadIdx.x; i < BK * d_prime; i += blockDim.x) {
            int m = t * BK + i / d_prime;
            if (m < M) {
                V_tile[i] = V[m * d_prime + (i % d_prime)];
            }
        }

        __syncthreads();

        // 计算注意力分数
        for (int k = 0; k < BK; k++) {
            int m = t * BK + k;
            if (m >= M) break;

            // 计算 Q·K
            float score = 0;
            for (int i = 0; i < d; i++) {
                score += q[i] * __half2float(K_tile[k * d + i]);
            }
            score /= sqrtf(d);

            // 在线 softmax
            float old_max = max_val;
            max_val = fmaxf(max_val, score);
            float exp_score = expf(score - max_val);
            sum_exp = sum_exp * expf(old_max - max_val) + exp_score;

            // 累加 V
            for (int i = 0; i < d_prime; i++) {
                acc[i] += exp_score * __half2float(V_tile[k * d_prime + i]);
            }
        }

        __syncthreads();
    }

    // 归一化并存储
    for (int i = 0; i < d_prime; i++) {
        O[n * d_prime + i] = acc[i] / sum_exp;
    }
}
```

#### 6.5.1 内存映射 (mmap)

```cpp
// src/llama-mmap.cpp

// 内存映射优势：
// 1. 减少启动时间（不需要加载整个模型到内存）
// 2. 支持懒加载（只加载实际访问的页面）
// 3. 允许操作系统智能管理内存
// 4. 多个进程可以共享同一模型的内存映射

struct llama_mmap {
    void * addr;
    size_t size;
    int fd;

    static llama_mmap from_file(const std::string & fname) {
        llama_mmap result;

        result.fd = open(fname.c_str(), O_RDONLY);
        if (result.fd < 0) {
            throw std::runtime_error("failed to open file");
        }

        result.size = lseek(result.fd, 0, SEEK_END);
        result.addr = mmap(nullptr, result.size, PROT_READ, MAP_PRIVATE, result.fd, 0);

        if (result.addr == MAP_FAILED) {
            close(result.fd);
            throw std::runtime_error("mmap failed");
        }

        // 使用 MADV_FREE 允许内核在内存压力下回收页面
        #ifdef MADV_FREE
            madvise(result.addr, result.size, MADV_FREE);
        #endif

        return result;
    }

    ~llama_mmap() {
        if (addr) munmap(addr, size);
        if (fd >= 0) close(fd);
    }
};
```

```cpp
// src/llama-kv-cache.cpp

// 连续批处理：动态添加/移除序列
class llama_kv_cache {
    // 找到空闲单元格用于新序列
    std::vector<int> find_free_cells(
        llama_seq_id seq_id,
        int n_tokens,
        DefragOption defrag)
    {
        std::vector<int> free_cells;

        // 查找空闲单元格
        for (int i = 0; i < cells.size() && free_cells.size() < n_tokens; i++) {
            if (cells[i].is_empty()) {
                free_cells.push_back(i);
            }
        }

        // 如果空间不足，触发碎片整理
        if (free_cells.size() < n_tokens && defrag != DefragOption::NONE) {
            defrag();
            // 重新查找
            free_cells.clear();
            for (int i = 0; i < cells.size() && free_cells.size() < n_tokens; i++) {
                if (cells[i].is_empty()) {
                    free_cells.push_back(i);
                }
            }
        }

        return free_cells;
    }

    // 碎片整理：压缩 KV 缓存
    void defrag() {
        // 1. 收集所有活跃单元格
        std::vector<std::pair<llama_pos, int>> active_cells;
        for (int i = 0; i < cells.size(); i++) {
            if (!cells[i].is_empty()) {
                active_cells.push_back({cells[i].pos, i});
            }
        }

        // 2. 按位置排序
        std::sort(active_cells.begin(), active_cells.end());

        // 3. 移动到连续位置
        std::vector<ggml_tensor_copy_plan> copy_plans;
        for (size_t i = 0; i < active_cells.size(); i++) {
            int src = active_cells[i].second;
            int dst = i;
            if (src != dst) {
                // 计划复制操作
                copy_plans.push_back({src, dst});
            }
        }

        // 4. 执行复制（批量）
        execute_copy_plans(copy_plans);

        // 5. 更新单元格状态
        update_cells_after_defrag();
    }
};
```

#### 6.6.2 批处理大小调优

```cpp
// common/arg.cpp

// 批处理大小对性能的影响
//
// n_batch: 逻辑批处理大小（一次处理的 token 数）
// n_ubatch: 物理批处理大小（实际分配的资源）
//
// 调优建议：
// - 小 n_batch: 低延迟，适合交互式应用
// - 大 n_batch: 高吞吐，适合批处理任务
// - n_ubatch < n_batch: 节省内存，但可能影响性能

struct common_params {
    uint32_t n_batch = 512;    // 默认批处理大小
    uint32_t n_ubatch = 512;   // 物理批处理大小

    // 自动调优（实验性）
    struct {
        bool enabled = false;
        size_t max_memory;  // 最大可用内存
        int target_latency; // 目标延迟（ms）
    } batch_tuning;

    void auto_tune_batch_size() {
        if (!batch_tuning.enabled) return;

        // 基于可用内存计算最大批处理大小
        size_t kv_cache_size = estimate_kv_cache_size();
        size_t compute_size = estimate_compute_size();

        size_t available = batch_tuning.max_memory - kv_cache_size;
        n_ubatch = available / compute_size;

        // 基于目标延迟调整
        if (batch_tuning.target_latency < 100) {
            n_batch = std::min(n_batch, n_ubatch / 2);  // 低延迟模式
        }

        LLAMA_LOG_INFO("Auto-tuned batch size: n_batch=%d, n_ubatch=%d\n",
                      n_batch, n_ubatch);
    }
};
```

### 6.7 推测解码

#### 6.7.1 推测解码原理

```cpp
// common/speculative.h

// 推测解码：使用小模型（draft）预测，大模型（target）验证
struct speculative_decoder {
    llama_context * draft_ctx;   // 小模型（快速）
    llama_context * target_ctx;  // 大模型（准确）

    int n_draft = 4;  // 每次预测的 token 数

    // 推测解码流程
    std::vector<llama_token> decode() {
        std::vector<llama_token> draft_tokens;
        std::vector<llama_token> accepted_tokens;

        // 1. 使用 draft 模型生成 n_draft 个 token
        for (int i = 0; i < n_draft; i++) {
            llama_token token = draft_model_sample();
            draft_tokens.push_back(token);
            draft_model_decode(token);
        }

        // 2. 使用 target 模型并行验证所有 draft tokens
        llama_batch batch = build_verification_batch(draft_tokens);
        target_model_decode(batch);

        // 3. 比较概率分布，决定是否接受
        for (int i = 0; i < draft_tokens.size(); i++) {
            float p_draft = draft_model.get_probability(draft_tokens[i]);
            float p_target = target_model.get_probability(draft_tokens[i]);

            if (rand() / RAND_MAX < p_target / p_draft) {
                // 接受
                accepted_tokens.push_back(draft_tokens[i]);
            } else {
                // 拒绝，从 target 分布重新采样
                llama_token new_token = target_model_resample();
                accepted_tokens.push_back(new_token);
                break;  // 停止验证
            }
        }

        return accepted_tokens;
    }
};
```

### 6.9 性能调优建议

#### 6.9.1 CPU 优化建议

```
1. 线程数设置：
   - n_threads: 设置为性能核数量
   - n_threads_batch: 可设置为总核心数

2. 内存带宽：
   - 使用多通道内存
   - 确保内存运行在标称频率

3. 量化选择：
   - Q4_K_M: 最佳平衡点
   - Q8_0: 质量优先
   - Q3_K_M: 速度优先

4. 批处理：
   - 交互式：n_batch = 1-8
   - 批处理：n_batch = 512-2048
```

#### 6.9.2 GPU 优化建议

```
1. VRAM 管理：
   - 确保模型完全放入 VRAM
   - 使用 --n-gpu-layers 调整卸载层数

2. 批处理：
   - GPU 适合大批处理：n_batch = 1024-4096
   - 调整 --tensor-split 多 GPU 分配

3. 内核优化：
   - CUDA: 调整 --cuda-flags
   - Metal: 使用 --metal-flags

4. Flash Attention：
   - 启用 --flash-attn 提高注意力计算速度
   - 需要足够的 VRAM
```

#### 6.9.3 内存优化建议

```
1. 使用 mmap:
   --mmap 启用内存映射

2. KV 缓存量化：
   --cache-type-k q8_0
   --cache-type-v q8_0

3. 批处理大小：
   根据可用内存调整 n_batch

4. 上下文大小：
   --ctx-size 仅设置需要的上下文
```
