### 2.1 模型加载逻辑

#### 2.1.1 GGUF 文件格式解析

GGUF (GGML Universal Format) 是 llama.cpp 使用的模型文件格式，支持：
- 元数据存储（架构、超参数、词汇表）
- 张量数据存储
- 量化权重存储

**解析流程：**

```cpp
llama_model_loader::llama_model_loader(const std::string & fname, ...) {
    // 1. 打开文件并映射到内存
    struct gguf_context * ctx_gguf = gguf_init_from_file(fname.c_str(), ...);

    // 2. 读取元数据
    - 模型架构 (architecture)
    - 超参数 (hyperparameters)
    - 词汇表 (vocab)
    - 量化信息 (quantization)

    // 3. 加载张量信息
    - 张量名称
    - 张量维度
    - 张量类型 (FP16, Q4_K, Q8_0, etc.)
    - 文件偏移量

    // 4. 创建内存映射
    - 使用 mmap 映射权重文件
    - 减少内存占用，支持懒加载
}
```

#### 2.2.1 llama_context 初始化

```cpp
llama_context::llama_context(
    const llama_model & model,
    llama_context_params params)
{
    // 1. 复制模型超参数到上下文
    m_hparams = model.hparams;
    m_cparams = params;

    // 2. 创建内存对象
    - llama_memory_i (KV 缓存或状态空间)
    - 根据模型类型选择：

      * RNN/状态空间模型：llama_memory_recurrent
      * 混合模型：llama_memory_hybrid

    // 3. 构建计算图
    - llama_graph (llama-graph.cpp)
    - 预留最坏情况下的图大小
    - 根据 n_batch, n_ubatch, n_ctx 计算

    // 4. 初始化后端调度器
    - ggml_backend_sched
    - 分配计算缓冲区
    - 设置线程池

    // 5. 初始化采样器链
    - llama_sampler_chain
    - 根据 params.sampling 配置
}
```

### 2.3 批处理逻辑

#### 2.3.1 llama_batch 结构

```cpp
// src/llama-batch.h
struct llama_batch {
    int32_t n_tokens;       // token 数量

    llama_token  * token;   // token 数据 [n_tokens]
    llama_pos    * pos;     // 位置信息 [n_tokens]
    int32_t      * n_seq_id; // 序列数量 [n_tokens]
    llama_seq_id ** seq_id; // 序列 ID [n_tokens][n_seq_id]
    int8_t       * logits;  // 是否需要 logits [n_tokens]

    // 多模态支持
    llama_embd   * embd;    // 嵌入输入 (可选)
};
```

#### 2.3.2 批处理调度

```cpp
// 批处理构建流程
void llama_context::decode(llama_batch batch) {
    // 1. 验证批处理参数
    GGML_ASSERT(batch.n_tokens <= n_batch());
    GGML_ASSERT(batch.n_tokens <= n_ubatch());

    // 2. 为每个序列分配 KV 缓存空间
    for (int i = 0; i < batch.n_tokens; i++) {
        llama_seq_id seq_id = batch.seq_id[i][0];
        kv_cache.allocate_tokens(seq_id, ...);
    }

    // 3. 构建计算图
    auto graph = m_graph.build_forward(..., batch, ...);

    // 4. 执行计算
    ggml_backend_sched_graph_compute(m_sched, graph);

    // 5. 更新 KV 缓存
    kv_cache.commit();
}
```

#### 2.4.3 语法约束采样

```cpp
// src/llama-grammar.cpp
class llama_grammar {
    // 语法规则解析（GBNF 格式）
    std::vector<llama_grammar_rule> rules;

    // 解析状态栈
    std::vector<std::vector<const llama_grammar_element *>> stacks;

    );

    // 接受一个 token，更新语法状态
    void accept_token(llama_token token);
};

// GBNF 语法示例：
// root ::= expression
// expression ::= term ("+" term)*
// term ::= [0-9]+
```

**语法约束流程：**

```cpp
llama_token common_sampler_sample(
    common_sampler * gsmpl,
    llama_context * ctx,
    int idx,
    bool grammar_first = false)
{
    // 1. 获取 logits
    float * logits = llama_get_logits_ith(ctx, idx);

    // 2. 设置候选 token
    candidates.data = ...;
    candidates.size = vocab_size;

    // 3. 应用采样器链
    llama_sampler_chain_apply(gsmpl->chain, logits, &candidates);

    // 4. 语法约束（如果启用）
    if (grammar) {
        // 过滤不符合语法的 token
        grammar->apply_constraints(&candidates, vocab);

        // 重新采样
        if (candidates.size == 0) {
            // 重置并重新应用约束
            grammar->reset();
        }
    }

    // 5. 采样最终 token
    llama_token token = sample_token(&candidates);

    // 6. 接受 token
    gsmpl->accept(token);
    grammar->accept_token(token);

    return token;
}
```

#### 2.5.2 KV 缓存操作

```cpp
// 分配 KV 缓存空间
bool llama_kv_cache::alloc(
    const llama_hparams & hparams,
    const llama_cparams & cparams)
{
    // 计算每层 KV 大小
    size_t k_size = hparams.n_embd_head_k * hparams.n_head_kv * n_ctx;
    size_t v_size = hparams.n_embd_head_v * hparams.n_head_kv * n_ctx;

    // 分配连续内存
    for (int l = 0; l < n_layer; l++) {
        k[l] = ggml_new_tensor_4d(..., type_k, ...);
        v[l] = ggml_new_tensor_4d(..., type_v, ...);
    }

    // 初始化单元格
    cells.resize(n_ctx);
}

// 查找空闲单元格
std::vector<int> llama_kv_cache::find_free_cells(
    const llama_batch & batch,
    int n_tokens)
{
    std::vector<int> free_cells;

    for (int i = 0; i < cells.size() && free_cells.size() < n_tokens; i++) {
        if (cells[i].is_empty()) {
            free_cells.push_back(i);
        }
    }

    // 如果空间不足，触发碎片整理或驱逐
    if (free_cells.size() < n_tokens) {
        defrag();  // 或 evict_oldest()
    }

    return free_cells;
}
```

#### 2.5.3 注意力计算中的 KV 使用

```cpp
// 注意力计算（简化版）
struct ggml_tensor * llama_graph::build_attention(
    int il,  // 层索引
    ggml_tensor * cur,
    ggml_tensor * inp_pos,
    ggml_tensor * inp_kq_mask)
{
    // 1. 计算 Q, K, V
    ggml_tensor * Q = ggml_mul_mat(ctx, wq, cur);  // [n_batch, n_embd]
    ggml_tensor * K = ggml_mul_mat(ctx, wk, cur);
    ggml_tensor * V = ggml_mul_mat(ctx, wv, cur);

    // 2. RoPE 旋转位置编码
    Q = ggml_rope(ctx, Q, inp_pos, rope_params);
    K = ggml_rope(ctx, K, inp_pos, rope_params);

    // 3. 存储 KV 到缓存
    ggml_build_forward_expand(graph, ggml_copy(ctx, K, k[il]));
    ggml_build_forward_expand(graph, ggml_copy(ctx, V, v[il]));

    // 4. 注意力计算
    ggml_tensor * KQ = ggml_mul_mat(ctx, K, Q);  // [n_batch, n_ctx]
    ggml_tensor * KQ_scaled = ggml_scale(ctx, KQ, scale_factor);
    ggml_tensor * KQ_masked = ggml_add(ctx, KQ_scaled, inp_kq_mask);
    ggml_tensor * KQ_soft_max = ggml_soft_max(ctx, KQ_masked);

    // 5. 输出
    ggml_tensor * KQV = ggml_mul_mat(ctx, V, KQ_soft_max);

    return KQV;
}
```

### 2.6 多线程与并行

#### 2.6.1 线程池架构

```cpp
// ggml/src/ggml-cpu/ggml-cpu.cpp
struct ggml_threadpool {
    std::vector<std::thread> workers;
    std::queue<std::function<void()>> tasks;
    std::mutex queue_mutex;
    std::condition_variable condition;
    bool stop = false;

    void execute(ggml_compute_params * params, ggml_tensor * op) {
        // 将计算任务分配到线程池
        for (int tid = 0; tid < n_threads; tid++) {
            params[tid].ith = tid;
            params[tid].nth = n_threads;
        }

        // 并行执行
        parallel_for(n_threads, [&](int tid) {
            op->compute_func(op, params[tid]);
        });
    }
};
```

#### 2.6.2 后端调度器

```cpp
// ggml/src/ggml-backend.cpp
struct ggml_backend_sched {
    // 多个后端
    std::vector<ggml_backend_t> backends;

    // 每个后端的缓冲区
    std::vector<ggml_backend_buffer_t> buffers;

    // 调度策略
    void schedule_graph(ggml_cgraph * graph) {
        // 1. 拓扑排序
        auto sorted_nodes = topological_sort(graph);

        // 2. 为每个节点分配后端
        for (auto node : sorted_nodes) {
            ggml_backend_t best_backend = find_best_backend(node);
            node->backend = best_backend;

            // 3. 分配内存
            if (!is_offloaded(node)) {
                node->buffer = buffers[best_backend->index];
            }
        }

        // 4. 插入同步点
        insert_sync_points(graph);
    }
};
```

```cpp
// tools/server/server.cpp
class server_context {
    // 专用推理线程
    std::thread inference_thread;

    // 任务队列
    server_queue task_queue;

    // 响应队列
    server_response response_queue;

            server_task task = task_queue.receive();

        }
```

#### 2.7.2 内存映射优化

```cpp
// src/llama-mmap.cpp
void llama_mmap::init(const std::string & fname) {
    // 1. 打开文件
    int fd = open(fname.c_str(), O_RDONLY);

    // 2. 获取文件大小
    size_t size = lseek(fd, 0, SEEK_END);

    // 3. 创建映射
    void * addr = mmap(nullptr, size, PROT_READ, MAP_PRIVATE, fd, 0);

    // 4. 设置 madvise 提示
    #ifdef MADV_FREE
        madvise(addr, size, MADV_FREE);  // 允许内核回收
    #endif

    // 5. 预读优化（可选）
    #ifdef POSIX_MADV_WILLNEED
        madvise(addr, size, POSIX_MADV_WILLNEED);  // 即将访问
    #endif
}
```
